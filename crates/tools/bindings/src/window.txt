// ── Pointer opt-ins ───────────────────────────────────────────────────────────
// The window declares what it wants delivered; reading what arrives belongs to
// `windows-ui`. `EnableMouseInPointer` does not suppress the legacy message stream —
// `DefWindowProc` generates it from the pointer message — so a window that opts in
// must handle the pointer messages and return 0.
    EnableMouseInPointer
    SetWindowFeedbackSetting
    FEEDBACK_TYPE

// ── Quality of service ────────────────────────────────────────────────────────

    SetThreadInformation

    PROCESS_POWER_THROTTLING_STATE

// ── Display colour capability ─────────────────────────────────────────────────
// Whether the display is SDR, wide-gamut or HDR, and at what luminance, is a property
// of the window's monitor — so it belongs to the window rather than to the renderer.
// A desktop app has no CoreWindow, so `DisplayInformation` is reached through its
// interop factory; `AdvancedColorInfoChanged` is the single signal that covers an HDR
// toggle, an SDR-white-level change and a monitor hop alike.
    Windows.Graphics.Display.DisplayInformation
    Windows.Graphics.Display.AdvancedColorInfo
    Windows.Graphics.Display.AdvancedColorKind
    IDisplayInformationStaticsInterop
